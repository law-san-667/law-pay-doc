import { Tabs, TabItem } from '@astrojs/starlight/components';

## Endpoint

```http
GET /api/v1/payments
```

**Authentication:** `X-API-Key` header

## Query parameters

| Parameter | Type | Default | Description |
|-----------|------|---------|-------------|
| `status` | string | — | Filter by status (`pending`, `paid`, `cancelled`) |
| `limit` | number | 50 | Number of results (max 100) |
| `offset` | number | 0 | Offset for pagination |

## Request example

<Tabs>
  <TabItem label="cURL">
  ```bash
  # All paid transactions, page 1
  curl "https://api.lawpay.sn/api/v1/payments?status=paid&limit=20&offset=0" \
    -H "X-API-Key: lp_live_xxxxxxxxxxxxxxxxxxxx"
  ```
  </TabItem>
  <TabItem label="JavaScript">
  ```javascript
  const params = new URLSearchParams({
    status: 'paid',
    limit: '20',
    offset: '0',
  });

  const response = await fetch(
    `https://api.lawpay.sn/api/v1/payments?${params}`,
    {
      headers: { 'X-API-Key': process.env.LAWPAY_API_KEY },
    }
  );

  const { data } = await response.json();
  console.log(`${data.total} total transactions`);
  console.log(data.transactions); // Array of transactions
  ```
  </TabItem>
</Tabs>

## Pagination

Use `offset` and `limit` to paginate results:

```javascript
// Page 1
const page1 = await fetchPayments({ limit: 20, offset: 0 });

// Page 2
const page2 = await fetchPayments({ limit: 20, offset: 20 });

// Total pages
const totalPages = Math.ceil(page1.data.total / 20);
```

## Errors

| HTTP Code | Error Code | Description |
|-----------|------------|-------------|
